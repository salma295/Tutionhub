// Temporary password-reset tool for local XAMPP development.
$allowedAddresses = ["127.0.0.1", "::1"];
$remoteAddress = $_SERVER["REMOTE_ADDR"] ?? "";

if (!in_array($remoteAddress, $allowedAddresses, true)) {
    http_response_code(403);
    exit("This tool can only be used from this computer.");
}

require_once "db.php";

$message = "";
$success = false;

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    $email = trim($_POST["email"] ?? "");
    $newPassword = $_POST["new_password"] ?? "";

    if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $message = "Please enter a valid tutor email.";
    } elseif (!is_string($newPassword) || strlen($newPassword) < 8) {
        $message = "Password must contain at least 8 characters.";
    } else {
        $passwordHash = password_hash($newPassword, PASSWORD_DEFAULT);

        $stmt = $conn->prepare(
            "UPDATE tutors SET password_hash = ? WHERE email = ?"
        );
        $stmt->execute([$passwordHash, $email]);

        if ($stmt->rowCount() > 0) {
            $message = "Tutor password updated successfully!";
            $success = true;
        } else {
            $check = $conn->prepare(
                "SELECT tutor_id FROM tutors WHERE email = ?"
            );
            $check->execute([$email]);

            if ($check->fetch()) {
                $message = "Password updated. You can now log in.";
                $success = true;
            } else {
                $message = "No tutor account was found with that email.";
            }
        }
    }
}
